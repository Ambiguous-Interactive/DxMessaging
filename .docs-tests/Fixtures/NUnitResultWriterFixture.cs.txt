namespace ResultWriterProbe
{
    using System;
    using System.Globalization;
    using System.IO;
    using System.Xml;
    using DxMessaging.Ci;
    using NUnit.Framework.Interfaces;
    using NUnit.Framework.Internal;

    public static class Probe
    {
        public static void NoOp() { }

        public static string Publication(string filePath, string mode)
        {
            TestSuite suite = new TestSuite("publication");
            TestSuiteResult result = new TestSuiteResult(suite);
            for (int index = 0; index < 2; ++index)
            {
                TestMethod test = new TestMethod(new MethodWrapper(typeof(Probe), nameof(NoOp)));
                suite.Add(test);
                TestCaseResult leaf;
                if ((mode == "throw-first" && index == 0) || (mode == "throw-last" && index == 1))
                {
                    leaf = new PublicationResult(
                        test,
                        () =>
                        {
                            throw new IOException("controlled serializer failure");
                        }
                    );
                }
                else if (mode == "publish-directory" && index == 1)
                {
                    leaf = new PublicationResult(test, () => Directory.CreateDirectory(filePath));
                }
                else
                {
                    leaf = mode == "custom" ? new CustomResult(test) : new TestCaseResult(test);
                }
                leaf.SetResult(ResultState.Success);
                result.AddResult(leaf);
            }

            int accelerated = DxmNUnitResultWriter.WriteFile(
                Header("publication"),
                result,
                filePath,
                mode == "throw-dispose"
                    ? (Func<Stream, Stream>)(output => new ClosingFailureStream(output))
                    : null
            );
            int expectedAccelerated = mode == "custom" ? 0 : 2;
            if (accelerated != expectedAccelerated)
            {
                return "Expected accelerated=" + expectedAccelerated + "; actual=" + accelerated;
            }
            using (MemoryStream stream = new MemoryStream())
            {
                using (StreamWriter text = new StreamWriter(stream))
                {
                    using (
                        XmlWriter writer = XmlWriter.Create(
                            text,
                            new XmlWriterSettings { Indent = true }
                        )
                    )
                    {
                        TNode run = Header("publication");
                        run.ChildNodes.Add(result.ToXml(true));
                        run.WriteTo(writer);
                    }
                    text.Flush();
                    byte[] expected = stream.ToArray();
                    byte[] actual = File.ReadAllBytes(filePath);
                    if (expected.Length != actual.Length)
                    {
                        return "Complete XML byte lengths differ";
                    }
                    for (int index = 0; index < expected.Length; ++index)
                    {
                        if (expected[index] != actual[index])
                        {
                            return "Complete XML bytes differ at " + index;
                        }
                    }
                }
            }
            return "equal";
        }

        public static string Compare(
            string payload,
            string status,
            string label,
            string site,
            string shape,
            string culture
        )
        {
            CultureInfo previous = CultureInfo.CurrentCulture;
            try
            {
                CultureInfo.CurrentCulture = CultureInfo.GetCultureInfo(culture);
                MethodWrapper method = new MethodWrapper(typeof(Probe), nameof(NoOp));
                TestMethod test =
                    shape == "custom-test" ? new CustomMethod(method) : new TestMethod(method);
                test.Name = "name:" + payload;
                test.FullName = "full:" + payload;
                test.RunState = RunState.Explicit;
                test.Seed = 123;
                test.Properties.Add("Category", payload ?? string.Empty);
                test.Properties.Add("Category", "second");
                test.Properties.Add("property:" + payload, payload ?? string.Empty);
                TestCaseResult leaf =
                    shape == "custom-result" ? new CustomResult(test) : new TestCaseResult(test);
                leaf.SetResult(
                    new ResultState(
                        (TestStatus)Enum.Parse(typeof(TestStatus), status),
                        label,
                        (FailureSite)Enum.Parse(typeof(FailureSite), site)
                    ),
                    payload,
                    payload
                );
                leaf.StartTime = new DateTime(2026, 11, 10, 1, 2, 3, DateTimeKind.Utc);
                leaf.EndTime = leaf.StartTime.AddSeconds(2);
                leaf.Duration = 1.23456789;
                leaf.OutWriter.Write(payload);
                TestSuite suite = new TestSuite("suite:" + payload);
                suite.Properties.Add("Category", payload ?? string.Empty);
                suite.Add(test);
                TestSuiteResult result =
                    shape == "custom-suite" ? new CustomSuite(suite) : new TestSuiteResult(suite);
                result.AddResult(leaf);
                result.StartTime = leaf.StartTime;
                result.EndTime = leaf.EndTime;
                result.Duration = leaf.Duration;
                result.OutWriter.Write(payload);
                if (shape == "suite-failure")
                {
                    result.SetResult(ResultState.Failure, payload, payload);
                }
                if (shape == "empty")
                {
                    result = new TestSuiteResult(new TestSuite("empty"));
                    result.SetResult(ResultState.Inconclusive, payload);
                }

                string expected;
                using (StringWriter text = new StringWriter(CultureInfo.InvariantCulture))
                {
                    using (
                        XmlWriter writer = XmlWriter.Create(
                            text,
                            new XmlWriterSettings { Indent = true }
                        )
                    )
                    {
                        TNode run = Header(payload);
                        run.ChildNodes.Add(result.ToXml(true));
                        run.WriteTo(writer);
                    }
                    expected = text.ToString();
                }
                string actual;
                int accelerated;
                using (StringWriter text = new StringWriter(CultureInfo.InvariantCulture))
                {
                    using (
                        XmlWriter writer = XmlWriter.Create(
                            text,
                            new XmlWriterSettings { Indent = true }
                        )
                    )
                    {
                        accelerated = DxmNUnitResultWriter.Write(Header(payload), result, writer);
                    }
                    actual = text.ToString();
                }
                int expectedAccelerated = shape == "standard" || shape == "suite-failure" ? 1 : 0;
                if (accelerated != expectedAccelerated)
                {
                    return "Expected accelerated leaves="
                        + expectedAccelerated
                        + "; actual="
                        + accelerated;
                }
                return expected == actual
                    ? "equal"
                    : "Expected:\n" + expected + "\nActual:\n" + actual;
            }
            finally
            {
                CultureInfo.CurrentCulture = previous;
            }
        }

        private static TNode Header(string value)
        {
            TNode node = new TNode("test-run");
            node.AddAttribute("extra", value);
            return node;
        }

        private sealed class CustomMethod : TestMethod
        {
            public CustomMethod(IMethodInfo method)
                : base(method) { }

            public override TNode AddToXml(TNode parent, bool recursive)
            {
                TNode node = base.AddToXml(parent, recursive);
                node.AddElementWithCDATA("custom-method", "method metadata");
                return node;
            }
        }

        private sealed class CustomResult : TestCaseResult
        {
            public CustomResult(TestMethod method)
                : base(method) { }

            public override TNode AddToXml(TNode parent, bool recursive)
            {
                TNode node = base.AddToXml(parent, recursive);
                node.AddElementWithCDATA("custom-result", "result metadata");
                return node;
            }
        }

        private sealed class ClosingFailureStream : Stream
        {
            private readonly Stream _inner;
            private bool _thrown;

            public ClosingFailureStream(Stream inner)
            {
                _inner = inner;
            }

            public override bool CanRead => _inner.CanRead;
            public override bool CanSeek => _inner.CanSeek;
            public override bool CanWrite => _inner.CanWrite;
            public override long Length => _inner.Length;
            public override long Position
            {
                get => _inner.Position;
                set => _inner.Position = value;
            }
            public override void Flush() => _inner.Flush();
            public override int Read(byte[] buffer, int offset, int count) =>
                _inner.Read(buffer, offset, count);
            public override long Seek(long offset, SeekOrigin origin) => _inner.Seek(offset, origin);
            public override void SetLength(long value) => _inner.SetLength(value);
            public override void Write(byte[] buffer, int offset, int count) =>
                _inner.Write(buffer, offset, count);

            protected override void Dispose(bool disposing)
            {
                try
                {
                    if (disposing)
                    {
                        _inner.Dispose();
                        if (!_thrown)
                        {
                            _thrown = true;
                            if (_inner.CanWrite)
                            {
                                throw new InvalidOperationException("The disposal control did not close its stream.");
                            }
                            throw new IOException("controlled stream disposal failure");
                        }
                    }
                }
                finally
                {
                    base.Dispose(disposing);
                }
            }
        }

        private sealed class PublicationResult : TestCaseResult
        {
            private readonly Action _serialize;

            public PublicationResult(TestMethod method, Action serialize)
                : base(method)
            {
                _serialize = serialize;
            }

            public override TNode AddToXml(TNode parent, bool recursive)
            {
                _serialize();
                return base.AddToXml(parent, recursive);
            }
        }

        private sealed class CustomSuite : TestSuiteResult
        {
            public CustomSuite(TestSuite suite)
                : base(suite) { }

            public override TNode AddToXml(TNode parent, bool recursive)
            {
                TNode node = base.AddToXml(parent, recursive);
                node.Value = "custom suite text";
                return node;
            }
        }
    }
}
